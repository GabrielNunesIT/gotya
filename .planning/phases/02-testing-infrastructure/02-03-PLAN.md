   ```go
   func TestCorpus(t *testing.T) {
       yangsDir := filepath.Join("assets", "yangs")
       loader := &corpusLoader{
           dir:         yangsDir,
           astCache:    make(map[string]*ast.Module),
           schemaCache: make(map[string]*schema.Module),
       }

       entries, err := os.ReadDir(yangsDir)
       if err != nil {
           t.Fatalf("read corpus dir %s: %v", yangsDir, err)
       }

       // Load all modules sequentially to avoid concurrent map access in corpusLoader.
       var modules []namedModule
       for _, entry := range entries {
           if entry.IsDir() {
               continue
           }
           name := entry.Name()
           if !strings.HasSuffix(name, ".yang") {
               continue
           }
           modName := strings.TrimSuffix(name, ".yang")

           schemaMod, loadErr := loader.Load(modName)
           if loadErr != nil {
               t.Logf("module %s: compiler errors (not a test failure): %v", modName, loadErr)
           }
           if schemaMod != nil {
               modules = append(modules, namedModule{name: modName, mod: schemaMod})
           }
       }

       if len(modules) == 0 {
           t.Fatal("corpus loaded zero modules — corpus directory may be missing or empty")
       }

       gen := golanggenerator.New(&golanggenerator.Options{
           PackageName: "corpus",
           RootName:    "Device",
       })

       // Run generation and format.Source assertions in parallel subtests.
       for _, nm := range modules {
           nm := nm // capture loop variable
           t.Run(nm.name, func(t *testing.T) {
               t.Parallel()
               var buf bytes.Buffer
               if genErr := gen.Generate(nm.mod, &buf); genErr != nil {
                   t.Errorf("Generate(%s): %v", nm.name, genErr)
                   return
               }
               if buf.Len() == 0 {
                   // Empty output is acceptable for modules with no generatable nodes.
                   return
               }
               if _, fmtErr := format.Source(buf.Bytes()); fmtErr != nil {
                   t.Errorf("Generate(%s) produced invalid Go syntax: %v", nm.name, fmtErr)
               }
           })
       }
   }
   ```

Key constraints to honor:
- corpusLoader is a copy (not import) of fileLoader — the //go:build ignore tag on
  generate.go prevents it from being imported.
- Compiler errors during Load() are logged with t.Logf, NOT t.Errorf. Real-world corpus
  files often have unresolvable cross-module imports. A compile error is not a test failure.
- format.Source is called per-module on Generate() output (not GenerateDevice). Generate()
  includes a package header so format.Source can parse it.
- Sequential load + parallel subtests avoids the data race on corpusLoader's map cache.
- If buf.Len() == 0 after Generate, skip the format check — empty output is valid for
  modules with no leaf/container/list nodes that the generator handles.
- Use t.Errorf (not t.Fatalf) inside subtests to collect all failures.

After writing the file, run `go build ./test/...` to confirm it compiles. Fix any import
path mismatches discovered by reading generate.go's actual imports.
